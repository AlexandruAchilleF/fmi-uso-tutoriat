4 · Directoare în C

readdir dă câte o intrare. NULL = gata.

exercitiu.c#include <dirent.h>
DIR *d = opendir("demo"); // NULL la eroare
struct dirent *e;
while ((e = readdir(d)) != NULL) {
if (e->d_name[0] == '.') continue;
printf("%lu %s\n", (unsigned long)e->d_ino, e->d_name);
}
closedir(d);
